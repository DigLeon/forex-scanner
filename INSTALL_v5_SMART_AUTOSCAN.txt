5. server.js — Smart Auto Scan import
--------------------------------------
Рядом с другими require добавь:

const {
    createSmartAutoScanGate
} = require('./smartAutoScanGate');

После let scanInProgress = false; добавь:

const smartAutoScanGate = createSmartAutoScanGate({
    settleMs: Number(process.env.SMART_AUTO_SCAN_SETTLE_MS) || 1800
});

function requestSmartAutoScan(meta = {}) {
    if (!autoScanEnabled) return false;

    const http = require('http');
    const params = autoScanQuery
        ? `${autoScanQuery}&auto=1&smart=1`
        : 'minScore=50&showWeak=1&auto=1&smart=1';

    console.log(
        '[SMART AUTO SCAN] request',
        meta.candleDatetime || '',
        '| dirty:',
        meta.dirty || {}
    );

    const request = http.get(
        `http://127.0.0.1:${PORT}/api/scan?${params}`,
        response => {
            response.resume();
            response.on('end', () => {
                console.log(
                    '[SMART AUTO SCAN] completed | status:',
                    response.statusCode
                );
            });
        }
    );

    request.on('error', error => {
        console.warn('[SMART AUTO SCAN ERROR]', error.message);
    });

    request.setTimeout(
        90 * 1000,
        () => request.destroy(new Error('Smart auto scan timeout'))
    );

    return true;
}

6. server.js — замени текущий onRealtimeClosed1mCandle
-------------------------------------------------------
Сейчас у тебя:

onRealtimeClosed1mCandle(({ symbol, candle, price }) => {
    const closePrice = Number(price ?? candle?.close);
    if (Number.isFinite(closePrice)) {
        evaluateRealtimeEntryWindow(symbol, closePrice, '1M CLOSE');
    }
});

Замени на:

onRealtimeClosed1mCandle(({ symbol, candle, price }) => {
    const closePrice = Number(price ?? candle?.close);

    if (Number.isFinite(closePrice)) {
        evaluateRealtimeEntryWindow(symbol, closePrice, '1M CLOSE');
    }

    if (!realtimeConfig.enabled) return;

    smartAutoScanGate.onClosed1m({
        symbol,
        candle,
        enabled: autoScanEnabled,
        scanInProgress: () => scanInProgress,
        requestScan: requestSmartAutoScan
    });
});

Это делает ОДИН полный scan на одну закрытую минуту,
а не один scan на каждую пару.

14. Почему M3/M1 не вырезаем из combinedAnalysis прямо сейчас
--------------------------------------------------------------
Текущий analysisEngine строит MTF локально из 1m history. Это не расходует REST.
Рефакторинг "не считать M3/M1 до candidate" изменит внутренний execution path.
До набора 100–200 сигналов это нежелательно, потому что мы хотим сохранить
алгоритм неизменённым и получить чистую статистическую выборку.

После 100–200 completed TRADE можно отдельным этапом оптимизировать CPU pipeline,
не смешивая это с текущим экспериментом.
